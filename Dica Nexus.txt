/** Som tático de abertura de Dossiê do Acervo Oficial */
export function playAcervoOpenSound(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  // 1. Chirp eletrônico de autenticação (440Hz -> 880Hz)
  const osc = ctx.createOscillator();
  const oscGain = ctx.createGain();
  osc.type = 'sine';
  osc.frequency.setValueAtTime(440, now);
  osc.frequency.exponentialRampToValueAtTime(880, now + 0.08);

  oscGain.gain.setValueAtTime(0.01, now);
  oscGain.gain.linearRampToValueAtTime(0.18, now + 0.02);
  oscGain.gain.exponentialRampToValueAtTime(0.001, now + 0.1);

  osc.connect(oscGain);
  oscGain.connect(ctx.destination);
  osc.start(now);
  osc.stop(now + 0.1);

  // 2. Micro-ruído de papel digital / arquivo desbloqueado
  const bufferSize = Math.floor(ctx.sampleRate * 0.05);
  const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
  const data = buffer.getChannelData(0);
  for (let i = 0; i < bufferSize; i++) {
    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.3));
  }
  const noise = ctx.createBufferSource();
  noise.buffer = buffer;
  const noiseGain = ctx.createGain();
  noiseGain.gain.setValueAtTime(0.08, now + 0.02);
  noiseGain.gain.exponentialRampToValueAtTime(0.001, now + 0.07);

  noise.connect(noiseGain);
  noiseGain.connect(ctx.destination);
  noise.start(now + 0.02);
}

### 🧠 3. MOTOR DE VINCULAÇÃO EM TEMPO REAL (`src/lib/acervo-matcher.ts`)

Algoritmo determinístico de pontuação que cruza semântica, bancas e tags do MestreCard contra o catálogo indexado:

```typescript
import { ACERVO_CATALOG, AcervoItem } from '../data/acervo-catalog';

export interface MatchedAcervoResult {
  item: AcervoItem;
  score: number;
  matchReason: string;
}

function normalize(text: string): string {
  return text
    .toLowerCase()
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .trim();
}

export function matchAcervoToCard(card: any, limit = 6): MatchedAcervoResult[] {
  if (!card) return [];

  const rawTokens = [
    card.topic || '',
    card.title || '',
    ...(card.sec01_header?.thematicAxes || []),
    ...(card.tags || []),
    card.concept || ''
  ].join(' ');

  const cardTokens = normalize(rawTokens)
    .split(/[\s,/:;.\-—]+/)
    .filter((t) => t.length >= 4);

  const results: MatchedAcervoResult[] = [];

  for (const item of ACERVO_CATALOG) {
    let score = 0;
    const reasons: string[] = [];

    const normFilename = normalize(item.filename);
    const normTitle = normalize(item.title);

    // 1. Correspondência exata em tópicos prioritários (+40 pts)
    for (const topic of item.topics) {
      const normTopic = normalize(topic);
      if (cardTokens.some((t) => normTopic.includes(t) || t.includes(normTopic))) {
        score += 40;
        reasons.push(`Tópico [${topic}]`);
        break;
      }
    }

    // 2. Correspondência no nome do arquivo / título (+30 pts)
    for (const token of cardTokens) {
      if (normFilename.includes(token) || normTitle.includes(token)) {
        score += 15;
      }
    }

    // 3. Bônus para Bancas de Elite (+25 pts se for Einstein, ENEM ou UNESP)
    if (item.banca === 'Albert Einstein') score += 25;
    if (item.banca === 'UNESP' || item.banca === 'ENEM') score += 20;

    // 4. Bônus de Categoria: Caderno de Exercícios (+35 pts) ou Gabarito Comentado (+30 pts)
    if (item.category === 'caderno_exercicios') score += 35;
    if (item.category === 'gabarito_comentado') score += 30;

    if (score > 45) {
      results.push({
        item,
        score,
        matchReason: reasons.slice(0, 2).join(' • ') || 'Relevância Temática'
      });
    }
  }

  return results.sort((a, b) => b.score - a.score).slice(0, limit);
}

```

function scan(dir, list = []) {
  if (!fs.existsSync(dir)) return list;
  for (const item of fs.readdirSync(dir, { withFileTypes: true })) {
    const full = path.join(dir, item.name);
    if (item.isDirectory()) scan(full, list);
    else if (item.name.toLowerCase().endsWith('.pdf') || item.name.toLowerCase().endsWith('.txt')) {
      list.push({ name: item.name, full, size: fs.statSync(full).size, rel: path.relative(ROOT_DIR, full) });
    }
  }
  return list;
}

const files = scan(ROOT_DIR);

function inferMetadata(file) {
  const p = file.rel.toLowerCase();
  let discipline = 'geral';
  if (p.includes('quimica')) discipline = 'quimica';
  else if (p.includes('biologia')) discipline = 'biologia';
  else if (p.includes('fisica')) discipline = 'fisica';
  else if (p.includes('matematica')) discipline = 'matematica';

  let category = 'prova_oficial';
  if (p.includes('gabarito') || p.includes('comentada') || p.includes('resolucao')) category = 'gabarito_comentado';
  else if (p.includes('caderno') || p.includes('lista') || p.includes('exercicios') || p.includes('extras')) category = 'caderno_exercicios';

  else if (p.includes('unesp')) banca = 'UNESP';
  else if (p.includes('uerj')) banca = 'UERJ';
  else if (p.includes('uece')) banca = 'UECE';
  else if (p.includes('ufg')) banca = 'UFG';

  const yearMatch = file.name.match(/20[12][0-9]/);
  const year = yearMatch ? parseInt(yearMatch[0], 10) : undefined;

  const topics = [];
