/** Corte de espada ou golpe de tese */
export function playSwordSlashSound(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  const bufferSize = ctx.sampleRate * 0.12;
  const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
  const data = buffer.getChannelData(0);
  for (let i = 0; i < bufferSize; i++) {
    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.2));
  }
  const noise = ctx.createBufferSource();
  noise.buffer = buffer;
  const filter = ctx.createBiquadFilter();
  filter.type = 'bandpass';
  filter.frequency.setValueAtTime(1800, now);
  filter.frequency.exponentialRampToValueAtTime(300, now + 0.12);

  const gain = ctx.createGain();
  gain.gain.setValueAtTime(0.35, now);
  gain.gain.exponentialRampToValueAtTime(0.001, now + 0.12);

  noise.connect(filter);
  filter.connect(gain);
  gain.connect(ctx.destination);
  noise.start(now);
}

/** Estalo metálico do Escudo Mnemônico defendendo sofisma */
export function playShieldBlockSound(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  const osc = ctx.createOscillator();
  const gain = ctx.createGain();
  osc.type = 'sine';
  osc.frequency.setValueAtTime(520, now);
  osc.frequency.exponentialRampToValueAtTime(180, now + 0.2);

  gain.gain.setValueAtTime(0.3, now);
  gain.gain.exponentialRampToValueAtTime(0.001, now + 0.2);

  osc.connect(gain);
  gain.connect(ctx.destination);
  osc.start(now);
  osc.stop(now + 0.2);
}

/** Fanfarra de Vitória Épica (+300 XP) */
export function playBossDefeatedFanfare(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  const notes = [261.63, 329.63, 392.00, 523.25]; // C E G C
  notes.forEach((freq, idx) => {
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    const delay = idx * 0.12;

    osc.type = 'triangle';
    osc.frequency.setValueAtTime(freq, now + delay);

    gain.gain.setValueAtTime(0.001, now + delay);
    gain.gain.linearRampToValueAtTime(0.25, now + delay + 0.02);
    gain.gain.exponentialRampToValueAtTime(0.0001, now + delay + 0.45);

    osc.connect(gain);
    gain.connect(ctx.destination);
    osc.start(now + delay);
    osc.stop(now + delay + 0.5);
  });
}

import React, { useState } from 'react';
import { initBossCombat, BossCombatState } from '../../lib/boss-engine';
import { playSwordSlashSound, playShieldBlockSound, playBossDefeatedFanfare, playCircuitShortSound } from '../../lib/audio';

interface LabSectionProps {
  card: any;
  soundEnabled?: boolean;
  onRewardXp?: (amount: number, label: string) => void;
}

export const LabSection: React.FC<LabSectionProps> = ({
  card,
  soundEnabled = true,
  onRewardXp

  const [mode, setMode] = useState<'proof' | 'roguelike'>('proof');
  const [selectedOption, setSelectedOption] = useState<number | null>(null);
  const [isAnswered, setIsAnswered] = useState<boolean>(false);
  const [combat, setCombat] = useState<BossCombatState>(initBossCombat());
  const [bossHitEffect, setBossHitEffect] = useState<boolean>(false);

  // Dados da questão do card
  const questionTitle = card?.labQuestion?.title || card?.title || 'Desafio Estrutural da Banca';
  const questionPrompt = card?.labQuestion?.prompt || card?.concept || 'Resolva o dilema tático proposto pela matriz conceitual.';
  const options = card?.labQuestion?.options || card?.options || [
    'Hipótese Primária: Validação causal direta com nexo estrutural íntegro.',
    'Sofisma Distrator Alfa: Correlação aparente desprovida de sustentação mecânica.',
    'Sofisma Distrator Beta: Inversão prematura de premissas com dispersão de carga.',
    'Sofisma Distrator Gama: Neutralização estática sem dissipação de fluxo.'
  ];
  const correctIndex = card?.labQuestion?.correctIndex ?? 0;

  const handleSelectOptionForProof = (idx: number) => {
    if (isAnswered) return;
    setSelectedOption(idx);
    setIsAnswered(true);
    if (idx === correctIndex && onRewardXp) {
      onRewardXp(100, '★ QUESTÃO DO LABORATÓRIO RESOLVIDA (+100 XP) ★');
    }
  };

  const handleRoguelikeAttack = (idx: number) => {
    if (combat.isVictory || combat.isDefeat) return;

    if (idx === correctIndex) {
      // CRITICAL HIT! Derrota o Boss instantaneamente
      playSwordSlashSound(soundEnabled);
      playBossDefeatedFanfare(soundEnabled);
      setBossHitEffect(true);
      setTimeout(() => setBossHitEffect(false), 500);

      setCombat(prev => ({
        ...prev,
        bossHp: 0,
        isVictory: true,
        combatLog: ['[COMBATE CRÍTICO]: Tese magistral desferida! O Examinador Implacável foi estilhaçado!', ...prev.combatLog]
      }));

      if (onRewardXp) {
        onRewardXp(300, '★ BOSS DEFEATED: EXAMINADOR IMPLACÁVEL DERROTADO (+300 XP) ★');
      }
    } else {
      // ERRO: Contra-ataque da banca
      playCircuitShortSound(soundEnabled);
      playShieldBlockSound(soundEnabled);
      const damage = 25;
      const nextHp = Math.max(0, combat.playerHp - damage);

      setCombat(prev => ({
        ...prev,
        playerHp: nextHp,
        isDefeat: nextHp === 0,
        bossRage: prev.bossRage + 1,
        combatLog: [`[CONTRA-ATAQUE]: Sofisma detectado! A banca contra-ataca (-${damage} HP).`, ...prev.combatLog]
      }));
    }
  };

  const handleUseMnemonicsShield = () => {
    if (combat.actionPoints <= 0 || combat.isVictory || combat.isDefeat) return;
    playShieldBlockSound(soundEnabled);

    // Encontra um distrator incorreto ainda não eliminado
    const wrongIndices = options
      .map((_, i) => i)
      .filter(i => i !== correctIndex && !combat.eliminatedDistractors.includes(options[i]));

    if (wrongIndices.length > 0) {
      const targetDistractor = options[wrongIndices[0]];
      setCombat(prev => ({
        ...prev,
        actionPoints: prev.actionPoints - 1,
        eliminatedDistractors: [...prev.eliminatedDistractors, targetDistractor],
        combatLog: [`[ESCUDO MNEMÔNICO]: Distrator vaporizado com sucesso! (-1 PA)`, ...prev.combatLog]
      }));
    }
  };

  return (
    <section className="relative rounded-2xl bg-slate-950 border border-slate-800 p-6 backdrop-blur-xl overflow-hidden shadow-2xl">
      {/* Cabeçalho com seletor de Modo */}
      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4 mb-6 border-b border-slate-800 pb-4">

          <h2 className="text-xl font-bold text-white tracking-tight flex items-center gap-2">
            Julgamento Final & Questão da Banca
            {combat.isVictory && <span className="text-xs px-2.5 py-0.5 rounded bg-emerald-500/20 text-emerald-300 border border-emerald-500/40">★ BOSS DERROTADO</span>}
          </h2>
        </div>

        <div className="flex p-1 bg-slate-900 border border-slate-800 rounded-xl">
          <button
            onClick={() => setMode('proof')}
            className={`px-3 py-1.5 rounded-lg text-xs font-semibold tracking-wide transition-all ${
              mode === 'proof' ? 'bg-slate-800 text-slate-100 shadow-sm' : 'text-slate-400 hover:text-slate-200'
            }`}
          >
            📝 Prova Escrita
          </button>
          <button
            onClick={() => setMode('roguelike')}
            className={`px-3 py-1.5 rounded-lg text-xs font-semibold tracking-wide transition-all flex items-center gap-1.5 ${
              mode === 'roguelike'
                ? 'bg-red-500/20 text-red-300 border border-red-500/40 shadow-[0_0_12px_rgba(239,68,68,0.25)]'
                : 'text-red-400 hover:text-red-300'
            }`}
          >
            ⚔️ Boss Fight Roguelike <span className="text-[10px] bg-red-500 text-slate-950 font-black px-1 rounded-sm">+300 XP</span>
          </button>
        </div>
      </div>

      {/* MODO PROVA ESCRITA (Padrão Canônico) */}
      {mode === 'proof' && (

            <h3 className="text-sm font-bold text-slate-200 mb-2">{questionTitle}</h3>
            <p className="text-xs text-slate-300 leading-relaxed">{questionPrompt}</p>
          </div>

          <div className="grid grid-cols-1 gap-2.5">
            {options.map((opt: string, idx: number) => {
              const isSelected = selectedOption === idx;
              const isCorrect = idx === correctIndex;
              let btnStyle = 'bg-slate-900/80 border-slate-800 text-slate-300 hover:border-slate-700';

              if (isAnswered) {
                if (isCorrect) btnStyle = 'bg-emerald-950/40 border-emerald-500/60 text-emerald-200';
                else if (isSelected) btnStyle = 'bg-red-950/40 border-red-500/60 text-red-200';
              }

              return (
                <button
                  key={idx}
                  onClick={() => handleSelectOptionForProof(idx)}
                  disabled={isAnswered}
                  className={`p-3.5 text-left rounded-xl border text-xs font-mono transition-all flex items-start gap-3 ${btnStyle}`}

// Adições táteis de combate RPG em src/lib/audio.ts

/** Corte de lâmina / Golpe de Tese preciso */
export function playSwordSlashSound(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  // Ruído com filtro passa-banda dinâmico em varredura rápida
  const bufferSize = Math.floor(ctx.sampleRate * 0.14);
  const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
  const data = buffer.getChannelData(0);
  for (let i = 0; i < bufferSize; i++) {
    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.22));
  }
  const noise = ctx.createBufferSource();
  noise.buffer = buffer;

  const filter = ctx.createBiquadFilter();
  filter.type = 'bandpass';
  filter.frequency.setValueAtTime(2200, now);
  filter.frequency.exponentialRampToValueAtTime(350, now + 0.14);
  filter.Q.setValueAtTime(3.5, now);

  const gain = ctx.createGain();
  gain.gain.setValueAtTime(0.35, now);
  gain.gain.exponentialRampToValueAtTime(0.001, now + 0.14);

  noise.connect(filter);
  filter.connect(gain);
  gain.connect(ctx.destination);
  noise.start(now);
}

/** Deflexão metálica do Escudo Mnemônico vaporizando o sofisma */
export function playShieldBlockSound(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  // Impacto metálico puro com ressonância harmônica cristalina
  const freqs = [620, 1240, 1860];
  freqs.forEach((freq, idx) => {
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = 'sine';
    osc.frequency.setValueAtTime(freq, now);
    osc.frequency.exponentialRampToValueAtTime(freq * 0.85, now + 0.25);

    gain.gain.setValueAtTime(0.2 / (idx + 1), now);
    gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.25);

    osc.connect(gain);
    gain.connect(ctx.destination);
    osc.start(now);
    osc.stop(now + 0.25);

}

/** Golpe cortante de tese (filtro passa-banda dinâmico) */
export function playSwordSlashSound(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  const bufferSize = Math.floor(ctx.sampleRate * 0.14);
  const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
  const data = buffer.getChannelData(0);
  for (let i = 0; i < bufferSize; i++) {
    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.2));
  }
  const noise = ctx.createBufferSource();
  noise.buffer = buffer;

  const filter = ctx.createBiquadFilter();
  filter.type = 'bandpass';
  filter.frequency.setValueAtTime(2400, now);
  filter.frequency.exponentialRampToValueAtTime(320, now + 0.14);
  filter.Q.setValueAtTime(3.5, now);

  const gain = ctx.createGain();
  gain.gain.setValueAtTime(0.35, now);
  gain.gain.exponentialRampToValueAtTime(0.001, now + 0.14);

  noise.connect(filter);
  filter.connect(gain);
  gain.connect(ctx.destination);
  noise.start(now);
}

/** Deflexão metálica do Escudo Mnemônico */
export function playShieldBlockSound(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  const freqs = [640, 1280, 1920];
  freqs.forEach((freq, idx) => {
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = 'sine';
    osc.frequency.setValueAtTime(freq, now);
    osc.frequency.exponentialRampToValueAtTime(freq * 0.8, now + 0.22);

    gain.gain.setValueAtTime(0.22 / (idx + 1), now);
    gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.22);

    osc.connect(gain);
    gain.connect(ctx.destination);
    osc.start(now);
    osc.stop(now + 0.22);
  });
}

/** Arpeggio do Oráculo de Foco */
export function playFocusOracleSound(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  const notes = [587.33, 739.99, 880.0, 1174.66]; // D5, F#5, A5, D6
  notes.forEach((freq, i) => {
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = 'sine';
    osc.frequency.setValueAtTime(freq, now + i * 0.05);

    gain.gain.setValueAtTime(0.001, now + i * 0.05);
    gain.gain.linearRampToValueAtTime(0.14, now + i * 0.05 + 0.01);
    gain.gain.exponentialRampToValueAtTime(0.0001, now + i * 0.05 + 0.32);

    osc.connect(gain);
    gain.connect(ctx.destination);
    osc.start(now + i * 0.05);
    osc.stop(now + i * 0.05 + 0.35);
  });
}

/** Rugido de Dano Crítico do Boss */
export function playBossHitRoarSound(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  const osc = ctx.createOscillator();
  const gain = ctx.createGain();
  osc.type = 'sawtooth';
  osc.frequency.setValueAtTime(95, now);
  osc.frequency.linearRampToValueAtTime(28, now + 0.3);

  const filter = ctx.createBiquadFilter();
  filter.type = 'lowpass';
  filter.frequency.setValueAtTime(420, now);
  filter.frequency.exponentialRampToValueAtTime(70, now + 0.3);

  gain.gain.setValueAtTime(0.4, now);
  gain.gain.exponentialRampToValueAtTime(0.001, now + 0.3);

  osc.connect(filter);
  filter.connect(gain);
  gain.connect(ctx.destination);

  osc.start(now);
  osc.stop(now + 0.3);
}

export function playBossDefeatedFanfare(enabled = true) {

  }, [card]);

  const correctIndex = card?.labQuestion?.correctIndex ?? 0;
  const oracleWeaknessClue = card?.coreEquation || card?.formula || card?.takeaway || 'Fraqueza detectada: A banca omite a invariância dimensional da grandeza na etapa intermediária.';

  // Gestão da trilha sonora ao alternar de modo ou desmontar
  useEffect(() => {
    if (mode === 'roguelike' && !combat.isVictory && !combat.isDefeat) {
      startBossBattleMusic(soundEnabled);
    } else {
      stopBossBattleMusic();
    }
    return () => {
      stopBossBattleMusic();
    };
  }, [mode, combat.isVictory, combat.isDefeat, soundEnabled]);

  // Ataque do Candidato
  const handleAttack = (idx: number) => {
    if (combat.isVictory || combat.isDefeat) return;

    if (idx === correctIndex) {
      // GOLPE CRÍTICO
      setScreenSlash(true);
      setBossShake(true);
      playSwordSlashSound(soundEnabled);
      playBossHitRoarSound(soundEnabled);

      if (typeof navigator !== 'undefined' && navigator.vibrate) {
        navigator.vibrate([30, 40, 60]);
      }

      setTimeout(() => {
        setBossShake(false);
        setScreenSlash(false);
        playBossVictoryFanfareAudio(soundEnabled);
      }, 400);

      setCombat((prev) => ({
        ...prev,
        bossHp: 0,
        isVictory: true,
        combatLog: [
          {
            id: `log-${Date.now()}`,
            type: 'crit',
            text: '★ GOLPE CRÍTICO DE TESE! A barreira de sofismas foi aniquilada com 1.000 de Dano!'
          },
          ...prev.combatLog
        ]
      }));

      if (onRewardXp) {
        onRewardXp(300, '★ EXAMINADOR IMPLACÁVEL DERROTADO (+300 XP) ★');
      }
    } else {
      // CONTRA-ATAQUE DA BANCA
      playShieldBlockSound(soundEnabled);
      const damage = 35 + combat.bossRageLevel * 5;
      const nextHp = Math.max(0, combat.playerHp - damage);
      const taunt = EXAMINER_TAUNTS[combat.bossRageLevel % EXAMINER_TAUNTS.length];

      if (typeof navigator !== 'undefined' && navigator.vibrate) {
        navigator.vibrate([80, 50, 80]);
      }

      setCombat((prev) => ({
        ...prev,
        playerHp: nextHp,
        isDefeat: nextHp === 0,
        bossRageLevel: prev.bossRageLevel + 1,
        turnCount: prev.turnCount + 1,
        combatLog: [
          {
            id: `log-${Date.now()}-boss`,
            type: 'boss',
            text: `⚔️ CONTRA-ATAQUE (-${damage} HP):${taunt}`
          },
          {
            id: `log-${Date.now()}-player`,
            type: 'player',
            text: `Alternativa [${String.fromCharCode(65 + idx)}] era um sofisma. Integridade comprometida.`
          },
          ...prev.combatLog
        ]
      }));
    }
  };

  // Escudo Mnemônico: Vaporiza 1 distrator (-1 PA)
  const handleUseMnemonicsShield = () => {
    if (combat.actionPoints <= 0 || combat.isVictory || combat.isDefeat) return;

    const availableDistractors = options
      .map((opt: string, i: number) => ({ opt, i }))
      .filter(({ opt, i }: { opt: string; i: number }) => i !== correctIndex && !combat.eliminatedDistractors.includes(opt));

    if (availableDistractors.length === 0) return;

    const target = availableDistractors[0];
    playShieldBlockSound(soundEnabled);

    setCombat((prev) => ({
      ...prev,
      actionPoints: prev.actionPoints - 1,
      eliminatedDistractors: [...prev.eliminatedDistractors, target.opt],
      combatLog: [
        {
          id: `log-${Date.now()}`,
          type: 'system',
          text: `🛡️ ESCUDO MNEMÔNICO: O sofisma [${String.fromCharCode(65 + target.i)}] foi desintegrado do tabuleiro! (-1 PA)`
        },
        ...prev.combatLog
      ]
    }));
  };

  // Oráculo de Foco: Revela ponto fraco da banca (-2 PA)
  const handleUseOracleSiphon = () => {
    if (combat.actionPoints < 2 || combat.oracleHintUsed || combat.isVictory || combat.isDefeat) return;
    playFocusOracleSound(soundEnabled);
    setShowOracleModal(true);

    setCombat((prev) => ({
      ...prev,
      actionPoints: prev.actionPoints - 2,
      oracleHintUsed: true,
      combatLog: [
        {
          id: `log-${Date.now()}`,
          type: 'system',
          text: '👁️ ORÁCULO DE FOCO: Vulnerabilidade epistêmica da banca descriptografada! (-2 PA)'
        },
        ...prev.combatLog
      ]
    }));
  };

  return (
    <section className="relative rounded-2xl bg-slate-950 border border-slate-800 p-6 backdrop-blur-xl overflow-hidden shadow-2xl">
      {/* Efeito Visual de Golpe de Tese em Tela Cheia */}
      {screenSlash && (
        <div className="absolute inset-0 z-50 pointer-events-none flex items-center justify-center">
          <div className="w-full h-1.5 bg-cyan-300 slash-effect" />
        </div>
      )}

      {/* Cabeçalho com comutador Prova Escrita vs Roguelike */}
      <div className="relative z-10 flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4 mb-6 border-b border-slate-800 pb-4">
        <div>
          <span className="text-[10px] font-mono uppercase tracking-widest text-cyan-400">SEÇÃO 05 // LABORATÓRIO TÁTICO</span>
          <h2 className="text-xl font-bold text-white tracking-tight flex items-center gap-2">
            Desafio Final & Arena do Cenáculo
            {combat.isVictory && (
              <span className="text-xs px-2.5 py-0.5 rounded-full bg-emerald-500/20 text-emerald-300 border border-emerald-400/40 font-mono shadow-[0_0_12px_#10b981]">
                ★ EXAMINADOR REFUTADO
              </span>
            )}
          </h2>
        </div>

        <div className="flex p-1 bg-slate-900 border border-slate-800 rounded-xl">
          <button
            onClick={() => setMode('proof')}
            className={`px-3 py-1.5 rounded-lg text-xs font-semibold tracking-wide transition-all ${
              mode === 'proof' ? 'bg-slate-800 text-slate-100 shadow-sm' : 'text-slate-400 hover:text-slate-200'
            }`}
          >
            📝 Prova Escrita
          </button>
          <button
            onClick={() => setMode('roguelike')}
            className={`px-3 py-1.5 rounded-lg text-xs font-semibold tracking-wide transition-all flex items-center gap-1.5 ${
              mode === 'roguelike'
                ? 'bg-red-500/20 text-red-300 border border-red-500/40 shadow-[0_0_12px_rgba(239,68,68,0.25)]'
                : 'text-red-400 hover:text-red-300'
            }`}
          >
            ⚔️ Boss Fight Roguelike <span className="text-[10px] bg-red-500 text-slate-950 font-black px-1 rounded-sm">+300 XP</span>
          </button>
        </div>
      </div>

      {/* 📝 MODO PROVA ESCRITA (CANÔNICO INTACTO) */}
      {mode === 'proof' && (
        <div className="space-y-4">
          <div className="p-4 rounded-xl bg-slate-900/60 border border-slate-800">
            <h3 className="text-sm font-bold text-slate-200 mb-2">{questionTitle}</h3>
            <p className="text-xs text-slate-300 leading-relaxed font-mono">{questionPrompt}</p>
          </div>

          <div className="grid grid-cols-1 gap-2.5">
            {options.map((opt: string, idx: number) => {
              const isSelected = selectedProofOption === idx;
              const isCorrect = idx === correctIndex;
              let btnClass = 'bg-slate-900/80 border-slate-800 text-slate-300 hover:border-slate-700';

              if (isProofAnswered) {
                if (isCorrect) btnClass = 'bg-emerald-950/50 border-emerald-500 text-emerald-200';
                else if (isSelected) btnClass = 'bg-red-950/50 border-red-500 text-red-200';
              }

              return (
                <button
                  key={idx}
                  onClick={() => {
                    if (isProofAnswered) return;
                    setSelectedProofOption(idx);
                    setIsProofAnswered(true);
                    if (idx === correctIndex && onRewardXp) {
                      onRewardXp(100, '★ QUESTÃO DE PROVA RESOLVIDA (+100 XP) ★');
                    }
                  }}
                  disabled={isProofAnswered}
                  className={`p-3.5 text-left rounded-xl border text-xs font-mono transition-all flex items-start gap-3 ${btnClass}`}
                >
                  <span className="font-bold px-2 py-0.5 rounded bg-slate-950 border border-slate-700">[{String.fromCharCode(65 + idx)}]</span>
                  <span className="flex-1 leading-relaxed">{opt}</span>
                </button>
              );
            })}
          </div>
        </div>
      )}

      {/* ⚔️ MODO BOSS FIGHT ROGUELIKE */}
      {mode === 'roguelike' && (
        <div className="relative space-y-6">
          {/* Background Texturizado da Arena com Vinheta */}
          <div 
            className="absolute inset-0 -m-6 pointer-events-none opacity-20 bg-cover bg-center rounded-2xl"
            style={{ backgroundImage: `url('/assets/boss/arena_bg.jpg')` }}
          />

          {/* PAINEL DO CHEFE: O Examinador Implacável */}
          <div className={`relative p-5 rounded-2xl bg-slate-900/90 border border-red-500/40 transition-all ${bossShake ? 'boss-damaged' : ''} ${combat.bossRageLevel > 0 ? 'boss-rage-active' : ''}`}>
            <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4 mb-3">
              <div className="flex items-center gap-3.5">
                {/* Avatar com imagem real fornecida e fallback estilizado */}
                <div className="relative w-14 h-14 rounded-xl overflow-hidden bg-slate-950 border-2 border-red-500/60 shadow-[0_0_15px_rgba(239,68,68,0.4)] flex-shrink-0">
                  <img
                    src="/assets/boss/examiner_avatar.jpg"
                    alt="O Examinador Implacável"
                    className="w-full h-full object-cover"
                    onError={(e) => {
                      (e.target as HTMLElement).style.display = 'none';
                    }}
                  />
                  <div className="absolute inset-0 flex items-center justify-center bg-gradient-to-br from-red-900 to-slate-950 text-red-400 font-bold -z-10">
                    👑
                  </div>
                </div>

                <div>
                  <div className="flex items-center gap-2">
                    <h4 className="text-base font-black text-white tracking-wide font-mono">O EXAMINADOR IMPLACÁVEL</h4>
                    <span className="text-[10px] font-mono px-2 py-0.5 rounded bg-red-950 text-red-400 border border-red-500/40">
                      NÍVEL 99
                    </span>
                  </div>
                  <span className="text-[11px] font-mono text-slate-400">
                    Fúria da Banca: Grau {combat.bossRageLevel} • Sofismas Ativos: {options.length - 1 - combat.eliminatedDistractors.length}
                  </span>
                </div>
              </div>

              <div className="text-right font-mono">
                <span className="text-xs text-red-400 uppercase font-bold">HP DO EXAMINADOR</span>
                <div className="text-2xl font-black text-red-300 drop-shadow-[0_0_8px_#ef4444]">
                  {combat.bossHp} <span className="text-xs text-slate-500 font-normal">/ 1.000</span>
                </div>
              </div>
            </div>

            {/* Barra Monolítica de HP */}
            <div className="w-full bg-slate-950 h-3.5 rounded-full overflow-hidden border border-slate-800 relative">
              <div
                className="h-full bg-gradient-to-r from-red-600 via-rose-500 to-red-400 transition-all duration-500 shadow-[0_0_12px_#ef4444]"
                style={{ width: `${(combat.bossHp / combat.maxBossHp) * 100}%` }}
              />
            </div>
          </div>

          {/* HUD DO CANDIDATO: Sanidade & Estamina */}
          <div className="relative grid grid-cols-1 md:grid-cols-12 gap-4 p-4 rounded-xl bg-slate-900/80 border border-slate-800">
            {/* Sanidade / Integridade */}
            <div className="md:col-span-6 space-y-1">
              <div className="flex justify-between text-xs font-mono">
                <span className="text-cyan-400 font-bold flex items-center gap-1.5">
                  🛡️ SANIDADE DO CANDIDATO
                </span>
                <span className="text-cyan-300 font-bold">{combat.playerHp} / {combat.maxPlayerHp} HP</span>
              </div>
              <div className="w-full bg-slate-950 h-2.5 rounded-full overflow-hidden border border-slate-800">
                <div
                  className="h-full bg-gradient-to-r from-cyan-500 to-blue-500 transition-all duration-300 shadow-[0_0_8px_#00f5ff]"
                  style={{ width: `${(combat.playerHp / combat.maxPlayerHp) * 100}%` }}
                />
              </div>
            </div>

            {/* Diamantes de Estamina TRI & Habilidades */}
            <div className="md:col-span-6 flex items-center justify-between md:justify-end gap-3">
              <div className="flex items-center gap-1.5">
                <span className="text-[10px] font-mono text-slate-400 mr-1">ESTAMINA:</span>
                {[0, 1, 2].map((i) => (
                  <div
                    key={i}
                    className={`w-3.5 h-3.5 rounded-sm transform rotate-45 border transition-all ${
                      i < combat.actionPoints
                        ? 'bg-cyan-400 border-cyan-300 shadow-[0_0_8px_#00f5ff]'
                        : 'bg-slate-900 border-slate-700 opacity-40'
                    }`}
                  />
                ))}
              </div>

              {!combat.isVictory && !combat.isDefeat && (
                <div className="flex gap-2">
                  <button
                    onClick={handleUseMnemonicsShield}
                    disabled={combat.actionPoints < 1}
                    className="px-2.5 py-1.5 rounded-lg bg-cyan-500/20 hover:bg-cyan-500/30 text-cyan-300 border border-cyan-500/40 text-[10px] font-mono font-bold transition-all disabled:opacity-30 disabled:cursor-not-allowed flex items-center gap-1"
                  >
                    🛡️ ESCUDO (-1 PA)
                  </button>

                  <button
                    onClick={handleUseOracleSiphon}
                    disabled={combat.actionPoints < 2 || combat.oracleHintUsed}
                    className="px-2.5 py-1.5 rounded-lg bg-amber-500/20 hover:bg-amber-500/30 text-amber-300 border border-amber-500/40 text-[10px] font-mono font-bold transition-all disabled:opacity-30 disabled:cursor-not-allowed flex items-center gap-1"
                  >
                    👁️ ORÁCULO (-2 PA)
                  </button>
                </div>
              )}
            </div>
          </div>

          {/* Enunciado do Confronto */}
          <div className="relative p-4 rounded-xl bg-slate-900/60 border border-slate-800">
            <span className="text-[10px] font-mono text-slate-400 uppercase tracking-widest">
              TURNO #{combat.turnCount} // ESCOLHA SEU GOLPE DE TESE
            </span>
            <h4 className="text-sm font-bold text-slate-200 mt-1 mb-1">{questionTitle}</h4>
            <p className="text-xs text-slate-300 leading-relaxed font-mono">{questionPrompt}</p>
          </div>

          {/* Opções de Golpe */}
          <div className="relative grid grid-cols-1 gap-3">
            {options.map((opt: string, idx: number) => {
              const isEliminated = combat.eliminatedDistractors.includes(opt);

              if (isEliminated) {
                return (
                  <div
                    key={idx}
                    className="p-3.5 rounded-xl border border-dashed border-slate-800 bg-slate-950/40 text-slate-600 text-xs font-mono line-through opacity-40 flex items-center gap-2"
                  >
                    <span>🛡️ [SOFISMA VAPORIZADO PELO ESCUDO]</span>
                    <span className="truncate">{opt}</span>
                  </div>
                );
              }

              return (
                <button
                  key={idx}
                  onClick={() => handleAttack(idx)}
                  disabled={combat.isVictory || combat.isDefeat}
                  className="group p-4 text-left rounded-xl bg-slate-900/90 border border-slate-800 hover:border-red-500/60 hover:shadow-[0_0_16px_rgba(239,68,68,0.25)] text-xs font-mono text-slate-300 hover:text-red-200 transition-all flex items-start gap-3 disabled:cursor-not-allowed"
                >
                  <span className="px-2 py-1 rounded bg-slate-950 border border-slate-700 text-red-400 text-[10px] font-bold group-hover:border-red-500">
                    🗡️ GOLPE [{String.fromCharCode(65 + idx)}]
                  </span>
                  <span className="flex-1 leading-relaxed">{opt}</span>
                  <span className="opacity-0 group-hover:opacity-100 text-red-400 text-xs transition-opacity font-bold">
                    [DESFERIR ➔]
                  </span>
                </button>
              );
            })}
          </div>

          {/* Modal do Oráculo */}
          {showOracleModal && (
            <div className="relative p-4 rounded-xl bg-amber-950/40 border border-amber-500/60 space-y-2">
              <div className="flex justify-between items-center">
                <span className="text-xs font-mono font-bold text-amber-300">📜 DESCRIPTOGRAFIA DO ORÁCULO:</span>
                <button
                  onClick={() => setShowOracleModal(false)}
                  className="text-amber-400 hover:text-amber-200 text-xs font-mono"
                >
                  [FECHAR]
                </button>
              </div>
              <p className="text-xs font-mono text-amber-200 leading-relaxed">{oracleWeaknessClue}</p>
            </div>
          )}

          {/* Estado de Vitória */}
          {combat.isVictory && (
            <div className="relative p-6 rounded-2xl bg-emerald-950/40 border border-emerald-500 text-center space-y-3 shadow-[0_0_30px_rgba(16,185,129,0.25)]">
              <div className="text-2xl font-black text-emerald-300 font-mono">
                ★ VITÓRIA ÉPICA: O EXAMINADOR FOI REFUTADO! ★
              </div>
              <p className="text-xs font-mono text-emerald-200 max-w-xl mx-auto leading-relaxed">
                Sua tese desarmou todos os sofismas da banca com precisão axiomática. 
                A recompensa de <strong>+300 XP</strong> foi transferida para o HUD principal.
              </p>
            </div>
          )}

          {/* Estado de Derrota */}
          {combat.isDefeat && (
            <div className="relative p-6 rounded-2xl bg-red-950/50 border border-red-500 text-center space-y-3 shadow-[0_0_30px_rgba(239,68,68,0.35)]">
              <div className="text-2xl font-black text-red-400 font-mono">
                ⚠️ SANIDADE ESGOTADA: VOCÊ FOI ANULADO PELA BANCA
              </div>
              <p className="text-xs font-mono text-red-200 max-w-xl mx-auto leading-relaxed">
                Os sofismas drenaram sua estamina. Rearme sua fundamentação na Prova Escrita ou recalibre seu raciocínio no Reator Causal.
              </p>
              <button
                onClick={() => setCombat(initBossCombat())}
                className="px-4 py-2 rounded-xl bg-red-600 hover:bg-red-500 text-white font-mono text-xs font-bold transition-all shadow-lg"
              >
                🔄 REINICIAR DUELO
              </button>
            </div>
          )}

          {/* Terminal de Logs de Batalha */}
          <div className="relative p-3.5 rounded-xl bg-slate-950/90 border border-slate-800 font-mono text-[11px] max-h-32 overflow-y-auto space-y-1.5 shadow-inner">
            {combat.combatLog.map((log) => (
              <div
                key={log.id}
                className={`${
                  log.type === 'crit'
                    ? 'text-emerald-400 font-bold'
                    : log.type === 'boss'
                    ? 'text-red-400'
                    : log.type === 'system'
                    ? 'text-cyan-300'
                    : 'text-slate-300'
                }`}
              >
                {log.text}
              </div>
            ))}
          </div>
        </div>
      )}
    </section>
  );
};
```
