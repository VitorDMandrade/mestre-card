### 🎵 2. ENGENHARIA ACÚSTICA: O SOM DE QUEBRA DE FASE (`src/lib/audio.ts`)

Sintetizador Web Audio API puro com estalo de cristal quebrando (ruído em alta frequência com ressonância) seguido pelo coice grave de queda de escudo mecânico:

```typescript
/** Quebra de Escudo de Fase do Boss (Shatter + Sub-Drop) */
export function playPhaseBreakSound(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  // 1. Estilhaço de Vidro/Cristal (Filtro Passa-Altas com decaimento exponencial)
  const bufferSize = Math.floor(ctx.sampleRate * 0.18);
  const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
  const data = buffer.getChannelData(0);
  for (let i = 0; i < bufferSize; i++) {
    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.15));
  }
  const noise = ctx.createBufferSource();
  noise.buffer = buffer;
  const highpass = ctx.createBiquadFilter();
  highpass.type = 'highpass';
  highpass.frequency.setValueAtTime(3200, now);

  const noiseGain = ctx.createGain();
  noiseGain.gain.setValueAtTime(0.3, now);
  noiseGain.gain.exponentialRampToValueAtTime(0.001, now + 0.18);

  noise.connect(highpass);
  highpass.connect(noiseGain);
  noiseGain.connect(ctx.destination);
  noise.start(now);

  // 2. Colapso Gravitacional de Fase (Varredura dente de serra de 520Hz a 80Hz)
  const osc = ctx.createOscillator();
  const oscGain = ctx.createGain();
  osc.type = 'sawtooth';
  osc.frequency.setValueAtTime(520, now);
  osc.frequency.exponentialRampToValueAtTime(80, now + 0.35);

  oscGain.gain.setValueAtTime(0.35, now);
  oscGain.gain.exponentialRampToValueAtTime(0.0001, now + 0.35);

  osc.connect(oscGain);
  oscGain.connect(ctx.destination);
  osc.start(now);
  osc.stop(now + 0.35);
}

```

### 📂 3. SCRIPT DE CATALOGAÇÃO INTELIGENTE DO ACERVO (`scripts/catalog-exams.mjs`)

Script nativo em Node.js (zero dependências) que varre recursivamente a pasta de exercícios, analisa nomes de arquivos, anos e formatos, categoriza por matéria/banca e gera o `CATALOGO_DO_ACERVO.md`:

```javascript
import fs from 'node:fs';
import path from 'node:path';

const ROOT_DIR = process.argv[2] || 'c:\\Users\\Vitor\\Exercícios e revisão\\Provas Oficiais e Exercícios';
const OUTPUT_FILE = path.join(process.cwd(), 'CATALOGO_DO_ACERVO.md');

function scanDirectory(dir, fileList = []) {
  if (!fs.existsSync(dir)) return fileList;
  const items = fs.readdirSync(dir, { withFileTypes: true });

  for (const item of items) {
    const fullPath = path.join(dir, item.name);
    if (item.isDirectory()) {
      scanDirectory(fullPath, fileList);
    } else {
      const ext = path.extname(item.name).toLowerCase();
      fileList.push({
        name: item.name,
        fullPath,
        ext,
        size: fs.statSync(fullPath).size,
        relPath: path.relative(ROOT_DIR, fullPath)
      });
    }
  }
  return fileList;
}

function generateCatalog() {
  console.log(`[CATALOGADOR]: Escaneando diretório: ${ROOT_DIR}...`);
  const files = scanDirectory(ROOT_DIR);
  const pdfs = files.filter(f => f.ext === '.pdf');
  const txts = files.filter(f => f.ext === '.txt');

  const groups = {
    einstein: pdfs.filter(f => f.relPath.toLowerCase().includes('einstein')),
    enem: pdfs.filter(f => f.relPath.toLowerCase().includes('enem')),
    unesp: pdfs.filter(f => f.relPath.toLowerCase().includes('unesp')),
    biologia: pdfs.filter(f => f.relPath.toLowerCase().includes('biologia')),
    quimica: pdfs.filter(f => f.relPath.toLowerCase().includes('quimica')),
    outros: []
  };

  let md = `# 🏛️ CATÁLOGO GERAL DO ACERVO DE PROVAS & EXERCÍCIOS\n\n`;
  md += `> **Data de Escaneamento**: ${new Date().toLocaleDateString('pt-BR')} | **Total de Arquivos**: ${files.length} (${pdfs.length} PDFs, ${txts.length} Textos)\n\n`;
  md += `## 📊 Resumo Executivo por Seção\n\n`;
  md += `| Categoria | Total de PDFs | Status de Ingestão |\n`;
  md += `| :--- | :--- | :--- |\n`;
  md += `| 🩺 **Medicina Albert Einstein** | ${groups.einstein.length} | Pronto para Injeção no Coliseu |\n`;
  md += `| 🌐 **Exame Nacional do Ensino Médio (ENEM)** | ${groups.enem.length} | Pronto para Injeção no Coliseu |\n`;
  md += `| 🏛️ **Vestibulares Paulistas (UNESP)** | ${groups.unesp.length} | Pronto para Injeção no Coliseu |\n`;
  md += `| 🧬 **Biologia Especializada (10 Tópicos)** | ${groups.biologia.length} | Mapeado para MestreCards |\n`;
  md += `| ⚗️ **Química Especializada (Tópicos)** | ${groups.quimica.length} | Mapeado para MestreCards |\n\n`;

  md += `## 📋 Inventário Detalhado dos Arquivos Oficiais\n\n`;
  md += `| Arquivo | Tamanho (KB) | Caminho Relativo |\n`;
  md += `| :--- | :--- | :--- |\n`;
  
  for (const file of pdfs.slice(0, 150)) {
    md += `| \`${file.name}\` | ${(file.size / 1024).toFixed(1)} KB | \`${file.relPath}\` |\n`;
  }

  if (pdfs.length > 150) {
    md += `| ... e mais ${pdfs.length - 150} arquivos catalogados no índice local. | - | - |\n`;
  }

  fs.writeFileSync(OUTPUT_FILE, md, 'utf-8');
  console.log(`[CATALOGADOR]: Sucesso! Catálogo gerado com ${files.length} itens em ${OUTPUT_FILE}`);
}

generateCatalog();
```

### 🧠 4. CLASSIFICADOR DE CONTEÚDO TÁTICO (`src/lib/question-classifier.ts`)

Mapeia deterministicamente termos técnicos para a disciplina e tópico corretos, permitindo que a arena selecione automaticamente o oponente ideal:

```typescript
export interface ClassifiedDomain {
  subject: 'química' | 'biologia' | 'física' | 'matemática' | 'geral';
  topic: string;
  recommendedBank: 'einstein' | 'enem' | 'unesp' | 'standard';
  confidence: number;
}

const TOPIC_DICTIONARY: Record<string, { subject: ClassifiedDomain['subject']; topic: string; defaultBank: ClassifiedDomain['recommendedBank'] }> = {
  // Química
  termoquímica: { subject: 'química', topic: 'Termoquímica & Entalpia', defaultBank: 'unesp' },
  eletroquímica: { subject: 'química', topic: 'Eletroquímica & Pilhas', defaultBank: 'enem' },
  estequiometria: { subject: 'química', topic: 'Cálculos Estequiométricos', defaultBank: 'unesp' },
  isomeria: { subject: 'química', topic: 'Química Orgânica & Isomeria', defaultBank: 'einstein' },
  // Biologia
  fisiologia: { subject: 'biologia', topic: 'Fisiologia Humana & Homeostase', defaultBank: 'einstein' },
  genética: { subject: 'biologia', topic: 'Genética & Biologia Molecular', defaultBank: 'einstein' },
  ecologia: { subject: 'biologia', topic: 'Ecologia & Ciclos Biogeoquímicos', defaultBank: 'enem' },
  citologia: { subject: 'biologia', topic: 'Citologia & Bioenergética', defaultBank: 'einstein' },
  // Física
  cinemática: { subject: 'física', topic: 'Mecânica & Cinemática', defaultBank: 'unesp' },
  óptica: { subject: 'física', topic: 'Óptica Geométrica', defaultBank: 'enem' }
};

export function classifyCardContent(title = '', prompt = '', tags: string[] = []): ClassifiedDomain {
  const combinedText = `${title} ${prompt} ${tags.join(' ')}`.toLowerCase();

  for (const [keyword, meta] of Object.entries(TOPIC_DICTIONARY)) {
    if (combinedText.includes(keyword)) {
      return {
        subject: meta.subject,
        topic: meta.topic,
        recommendedBank: meta.defaultBank,
        confidence: 0.95
      };
    }
  }

  return {
    subject: 'geral',
    topic: 'Fundamentos Interdisciplinares',
    recommendedBank: 'standard',
    confidence: 0.6
  };
}
```

```typescript
  recommendedBank: 'einstein' | 'enem' | 'unesp' | 'standard';
  confidence: number;
}

const TOPIC_MAP: Record<string, { subject: ClassifiedDomain['subject']; topic: string; defaultBank: ClassifiedDomain['recommendedBank'] }> = {
  termoquímica: { subject: 'química', topic: 'Termoquímica & Calorimetria', defaultBank: 'unesp' },
  eletroquímica: { subject: 'química', topic: 'Eletroquímica & Potencial Padrão', defaultBank: 'enem' },
  estequiometria: { subject: 'química', topic: 'Cálculo Estequiométrico', defaultBank: 'unesp' },
  isomeria: { subject: 'química', topic: 'Isomeria Espacial & Óptica', defaultBank: 'einstein' },
  fisiologia: { subject: 'biologia', topic: 'Fisiologia Humana & Regulação', defaultBank: 'einstein' },
  genética: { subject: 'biologia', topic: 'Genética & Engenharia Molecular', defaultBank: 'einstein' },
  ecologia: { subject: 'biologia', topic: 'Ecologia & Impactos Ambientais', defaultBank: 'enem' },
  citologia: { subject: 'biologia', topic: 'Citologia & Membrana Plasmática', defaultBank: 'einstein' }
};

export function classifyCardContent(title = '', prompt = '', tags: string[] = []): ClassifiedDomain {
  const text = `${title} ${prompt}${tags.join(' ')}`.toLowerCase();
  for (const [key, val] of Object.entries(TOPIC_MAP)) {
    if (text.includes(key)) {
      return { subject: val.subject, topic: val.topic, recommendedBank: val.defaultBank, confidence: 0.95 };
    }
  }
  return { subject: 'geral', topic: 'Domínio Interdisciplinar', recommendedBank: 'standard', confidence: 0.6 };
}
```

```typescript
/** Quebra de Escudo de Fase do Boss */
export function playPhaseBreakSound(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  // Estilhaço de Vidro/Cristal
  const bufferSize = Math.floor(ctx.sampleRate * 0.18);
  const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
  const data = buffer.getChannelData(0);
  for (let i = 0; i < bufferSize; i++) {
    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.15));
  }
  const noise = ctx.createBufferSource();
  noise.buffer = buffer;
  const highpass = ctx.createBiquadFilter();
  highpass.type = 'highpass';
  highpass.frequency.setValueAtTime(3200, now);

  const noiseGain = ctx.createGain();
  noiseGain.gain.setValueAtTime(0.3, now);
  noiseGain.gain.exponentialRampToValueAtTime(0.001, now + 0.18);

  noise.connect(highpass);
  highpass.connect(noiseGain);
  noiseGain.connect(ctx.destination);
  noise.start(now);

  // Colapso de Onda Dente de Serra
  const osc = ctx.createOscillator();
  const oscGain = ctx.createGain();
  osc.type = 'sawtooth';
  osc.frequency.setValueAtTime(520, now);
  osc.frequency.exponentialRampToValueAtTime(80, now + 0.35);

  oscGain.gain.setValueAtTime(0.35, now);
  oscGain.gain.exponentialRampToValueAtTime(0.0001, now + 0.35);

  osc.connect(oscGain);
  oscGain.connect(ctx.destination);
  osc.start(now);
  osc.stop(now + 0.35);
}
```

              <div className="absolute top-0 bottom-0 left-1/3 w-0.5 bg-slate-900/90 pointer-events-none" />
              <div className="absolute top-0 bottom-0 left-2/3 w-0.5 bg-slate-900/90 pointer-events-none" />
            </div>
          </div>

          {/* HUD DO CANDIDATO */}
          <div className="grid grid-cols-1 md:grid-cols-12 gap-4 p-4 rounded-xl bg-slate-900/80 border border-slate-800">
            <div className="md:col-span-6 space-y-1">
              <div className="flex justify-between text-xs font-mono">
                <span className="text-cyan-400 font-bold">🛡️ SANIDADE DO CANDIDATO</span>
                <span className="text-cyan-300 font-bold">{combat.playerHp} / {combat.maxPlayerHp} HP</span>
              </div>
              <div className="w-full bg-slate-950 h-2.5 rounded-full overflow-hidden border border-slate-800">
                <div
                  className="h-full bg-gradient-to-r from-cyan-500 to-blue-500 transition-all duration-300 shadow-[0_0_8px_#00f5ff]"
                  style={{ width: `${(combat.playerHp / combat.maxPlayerHp) * 100}%` }}
                />
              </div>
            </div>

            <div className="md:col-span-6 flex items-center justify-between md:justify-end gap-3">
              <div className="flex items-center gap-1.5">
                <span className="text-[10px] font-mono text-slate-400 mr-1">ESTAMINA:</span>
                {[0, 1, 2].map(i => (
                  <div
                    key={i}
                    className={`w-3.5 h-3.5 rounded-sm transform rotate-45 border transition-all ${
                      i < combat.actionPoints ? 'bg-cyan-400 border-cyan-300 shadow-[0_0_8px_#00f5ff]' : 'bg-slate-900 border-slate-700 opacity-40'
                    }`}
                  />
                ))}
              </div>

              {!combat.isVictory && !combat.isDefeat && (
                <div className="flex gap-2">
                  <button
                    onClick={handleUseMnemonicsShield}
                    disabled={combat.actionPoints < 1}
                    className="px-2.5 py-1.5 rounded-lg bg-cyan-500/20 hover:bg-cyan-500/30 text-cyan-300 border border-cyan-500/40 text-[10px] font-mono font-bold transition-all disabled:opacity-30 disabled:cursor-not-allowed"
                  >
                    🛡️ ESCUDO (-1 PA)
                  </button>
                  <button
                    onClick={handleUseOracleSiphon}
                    disabled={combat.actionPoints < 2 || combat.oracleHintUsed}
                    className="px-2.5 py-1.5 rounded-lg bg-amber-500/20 hover:bg-amber-500/30 text-amber-300 border border-amber-500/40 text-[10px] font-mono font-bold transition-all disabled:opacity-30 disabled:cursor-not-allowed"
                  >
                    👁️ ORÁCULO (-2 PA)
                  </button>
                </div>
              )}
            </div>
          </div>

          {/* INDICADOR DE FASE & ENUNCIADO DO ROUND */}
          <div className="p-4 rounded-xl bg-slate-900/60 border border-slate-800">
            <div className="flex items-center justify-between mb-1.5">
              <span className="text-xs font-mono font-bold text-amber-400 tracking-wider">
                {currentRound.title}
              </span>
              <span className="text-[10px] font-mono px-2 py-0.5 rounded bg-slate-950 text-slate-400 border border-slate-800">
                FASE {combat.currentRoundIndex + 1} DE {combat.totalRounds}
              </span>
            </div>
            <p className="text-xs text-slate-300 leading-relaxed font-mono">{currentRound.prompt}</p>
          </div>

          {/* ALTERNATIVAS DE GOLPE */}
          <div className="grid grid-cols-1 gap-3">
            {currentRound.options.map((opt: string, idx: number) => {
              const isEliminated = combat.eliminatedDistractors.includes(opt);

              if (isEliminated) {
                return (
                  <div key={idx} className="p-3.5 rounded-xl border border-dashed border-slate-800 bg-slate-950/40 text-slate-600 text-xs font-mono line-through opacity-40 flex items-center gap-2">
                    <span>🛡️ [SOFISMA VAPORIZADO]</span>
                    <span className="truncate">{opt}</span>
                  </div>
                );
              }

              return (
                <button
                  key={idx}
                  onClick={() => handleAttack(idx)}
                  disabled={combat.isVictory || combat.isDefeat || combat.isPhaseTransition}
                  className="group p-4 text-left rounded-xl bg-slate-900/90 border border-slate-800 hover:border-red-500/60 hover:shadow-[0_0_16px_rgba(239,68,68,0.25)] text-xs font-mono text-slate-300 hover:text-red-200 transition-all flex items-start gap-3 disabled:cursor-not-allowed"
                >
                  <span className="px-2 py-1 rounded bg-slate-950 border border-slate-700 text-red-400 text-[10px] font-bold group-hover:border-red-500">
                    🗡️ GOLPE [{String.fromCharCode(65 + idx)}]
                  </span>
                  <span className="flex-1 leading-relaxed">{opt}</span>
                  <span className="opacity-0 group-hover:opacity-100 text-red-400 text-xs transition-opacity font-bold">
                    [DESFERIR ➔]
                  </span>
                </button>
              );
            })}
          </div>

          {/* MODAL DO ORÁCULO */}
          {showOracleModal && (
            <div className="p-4 rounded-xl bg-amber-950/40 border border-amber-500/60 space-y-2">
              <div className="flex justify-between items-center">
                <span className="text-xs font-mono font-bold text-amber-300">📜 DESCRIPTOGRAFIA DA BANCA:</span>
                <button onClick={() => setShowOracleModal(false)} className="text-amber-400 hover:text-amber-200 text-xs font-mono">
                  [FECHAR]
                </button>
              </div>
              <p className="text-xs font-mono text-amber-200 leading-relaxed">{currentRound.clue}</p>
            </div>
          )}

          {/* TELAS DE FIM DE COMBATE */}
          {combat.isVictory && (
            <div className="p-6 rounded-2xl bg-emerald-950/40 border border-emerald-500 text-center space-y-3 shadow-[0_0_30px_rgba(16,185,129,0.25)]">
              <div className="text-2xl font-black text-emerald-300 font-mono">
                ★ VITÓRIA ÉPICA: {combat.selectedIdentity.badge} REFUTADO! ★
              </div>
              <p className="text-xs font-mono text-emerald-200 max-w-xl mx-auto leading-relaxed">
                Você superou as 3 fases do Gauntlet com rigor metodológico absoluto. A recompensa de <strong>+300 XP</strong> foi transferida para o HUD.
              </p>
            </div>
          )}

          {combat.isDefeat && (
            <div className="p-6 rounded-2xl bg-red-950/50 border border-red-500 text-center space-y-3 shadow-[0_0_30px_rgba(239,68,68,0.35)]">
              <div className="text-2xl font-black text-red-400 font-mono">
                ⚠️ SANIDADE ESGOTADA: O TRIBUNAL DA BANCA INDEFERIU SUA TESE
              </div>
              <button
                onClick={() => setCombat(initGauntletCombat(combat.selectedIdentity.id))}
                className="px-4 py-2 rounded-xl bg-red-600 hover:bg-red-500 text-white font-mono text-xs font-bold transition-all shadow-lg"
              >
                🔄 REINICIAR COLISEU
              </button>
            </div>
          )}

          {/* LOGS DE BATALHA */}
          <div className="p-3.5 rounded-xl bg-slate-950/90 border border-slate-800 font-mono text-[11px] max-h-32 overflow-y-auto space-y-1.5 shadow-inner">
            {combat.combatLog.map(log => (
              <div
                key={log.id}
                className={`${
                  log.type === 'crit'
                    ? 'text-emerald-400 font-bold'
                    : log.type === 'boss'
                    ? 'text-red-400'
                    : log.type === 'system'
                    ? 'text-cyan-300'
                    : 'text-slate-300'
                }`}
              >
                {log.text}
              </div>
            ))}
          </div>
        </div>
      )}
    </section>
  );
};
