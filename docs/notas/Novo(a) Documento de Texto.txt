          <h3 class="text-xl font-black text-white">Desafio Contra o Relógio</h3>
          <p class="text-sm text-slate-300 max-w-md mx-auto leading-relaxed">
            Você terá <strong class="text-red-400 font-mono">15 segundos</strong> por afirmação para responder <strong class="text-emerald-400 font-mono">Verdadeiro [V]</strong> ou <strong class="text-red-400 font-mono">Falso [F]</strong>.
          </p>
          <button onclick="renderG3()" class="mt-2 bg-gradient-to-r from-red-600 to-rose-600 hover:from-red-500 hover:to-rose-500 text-white font-black px-6 py-3 rounded-xl text-xs font-mono tracking-wider uppercase transition-all shadow-lg shadow-red-600/30 shortcut-next">
            INICIAR PRESSÃO TRI ➔ (Espaço)
          </button>
        </div>
      `;
    }

    function renderG3() {
      if(blockTimer) { clearInterval(blockTimer); blockTimer = null; }
      if(step3 >= tfData.length) { 
        gamesStatus.g3 = true; updateHUD(); calcFinalScore(); 
        if(timer) { clearInterval(timer); timer = null; }
        return document.getElementById('tf-container').innerHTML = `<div class="text-center p-6"><div class="text-5xl mb-3">🛡️</div><div class="text-blue-400 font-black text-xl">PROVA CONCLUÍDA!</div></div>`; 
      }
      const q = tfData[step3]; 
      if(timer) clearInterval(timer); 
      timeLeft = 15;
      const tText = document.getElementById('timer-text');
      const tBar = document.getElementById('timer-bar');
      tText.innerText = timeLeft + 's'; 
      tBar.style.width = '100%';
      document.getElementById('tf-container').innerHTML = `
        <div class="flex justify-between items-center mb-3">
          <span class="text-xs font-bold text-red-400 bg-red-400/10 px-2 py-1 rounded">Afirmação ${step3 + 1}/${tfData.length}</span>
        </div>
        <h3 class="text-lg text-white font-bold mb-6 text-center leading-relaxed">${q.statement}</h3>
        <div class="flex gap-4" id="tf-btns">
          <button id="btn-t" class="flex-1 bg-emerald-600 hover:bg-emerald-500 text-white font-black py-4 rounded-xl text-lg transition-colors">VERDADEIRO [V]</button>
          <button id="btn-f" class="flex-1 bg-red-600 hover:bg-red-500 text-white font-black py-4 rounded-xl text-lg transition-colors">FALSO [F]</button>
        </div>
        <div id="fb3" class="hidden mt-4 p-4 rounded-lg font-bold text-sm"></div>
      `;
      document.getElementById('btn-t').onclick = () => checkG3(true, q.isTrue, q.feedback);
      document.getElementById('btn-f').onclick = () => checkG3(false, q.isTrue, q.feedback);

      timer = setInterval(() => {
        timeLeft--; 
        tText.innerText = timeLeft + 's'; 
        tBar.style.width = (timeLeft/15)*100 + '%';
        if(timeLeft <= 0) { 
          clearInterval(timer); timer = null;
          checkG3(null, q.isTrue, "TEMPO ESGOTADO!"); 
        }
      }, 1000);
    }

    function checkG3(answer, isTrue, fb) {
      if(timer) { clearInterval(timer); timer = null; }
      if(answer === isTrue) scoreG3TimeSaved += timeLeft;
      const btnT = document.getElementById('btn-t');
      const btnF = document.getElementById('btn-f');
      if(btnT) btnT.disabled = true; 
      if(btnF) btnF.disabled = true;
      const el = document.getElementById('fb3'); 
      el.classList.remove('hidden');
      playSound(answer === isTrue);
      if(answer === isTrue) { 
        el.className = 'mt-4 p-4 rounded-lg font-bold text-sm bg-emerald-900/60 text-emerald-300 border border-emerald-500/30'; 
        el.innerText = `✅ CORRETO! ${fb}`;
        const nxt = document.createElement('button'); 
        nxt.className = 'block mt-3 bg-emerald-600 hover:bg-emerald-500 px-4 py-2 rounded text-white text-xs transition-colors shortcut-next'; 
        nxt.innerText = 'Próxima ➔ (Espaço)'; 
        nxt.onclick = () => { step3++; renderG3(); };
        el.appendChild(nxt);
      } else { 
        el.className = 'mt-4 p-4 rounded-lg font-bold text-sm bg-red-900/60 text-red-300 border border-red-500/30'; 
        el.innerText = `❌ ERRO! ${fb}`;
        applyDamage(20);
        triggerPunishment('tf-container', 5, () => { step3++; renderG3(); });
      }
      saveCardProgress();
    }

    let step4Seq = 0, step4Node = 0, orderShuffled = [];
    function initG4() { 
      if(step4Seq >= orderData.length) { gamesStatus.g4 = true; updateHUD(); calcFinalScore(); return document.getElementById('g4-container').innerHTML = `<div class="text-center p-6"><div class="text-5xl mb-3">🏆</div><div class="text-amber-400 font-black text-xl">TODAS AS SEQUÊNCIAS CONCLUÍDAS!</div></div>`; }
      totalAttemptsG4++;
      orderShuffled = shuffle(orderData[step4Seq].steps.map((txt, i) => ({txt, id: i, clicked: false}))); 
      step4Node = 0; renderG4(); 
    }
    function renderG4() {
      if(step4Node >= orderData[step4Seq].steps.length) {
        step4Seq++; 
        setTimeout(initG4, 1500);
        document.getElementById('g4-container').innerHTML = `<div class="text-center p-6 text-emerald-400 font-black text-xl">Sequência Concluída! Carregando próxima...</div>`;
        return;
      }
      let html = `<div class="flex justify-between items-center mb-3"><span class="text-xs font-bold text-amber-400 bg-amber-400/10 px-2 py-1 rounded">Sequência ${step4Seq + 1}/${orderData.length}</span></div><p class="text-sm text-gray-300 mb-2 font-bold">${orderData[step4Seq].title}</p><p class="text-xs text-gray-500 mb-4">Clique nos passos na ordem exata. Erros resetam a sequência.</p><div class="flex flex-col gap-2">`;
      orderShuffled.forEach((o) => { if(!o.clicked) html += `<button onclick="clickG4(${o.id})" class="p-3 bg-slate-800 border border-slate-700 rounded-lg text-sm text-left hover:bg-slate-700 text-gray-200 transition-colors">${o.txt}</button>`; });
      html += `</div><div class="mt-6"><div class="text-xs text-amber-400 font-bold mb-2 uppercase tracking-wider">Sua Sequência:</div><div class="flex flex-col gap-2">`;
      for(let i=0; i<step4Node; i++) { html += `<div class="p-3 bg-emerald-900/40 border border-emerald-500/30 rounded-lg text-sm text-emerald-300">✅ ${orderData[step4Seq].steps[i]}</div>`; }
      document.getElementById('g4-container').innerHTML = html + `</div></div>`;
      saveCardProgress();
    }
    function clickG4(id) {
      if(id === step4Node) { playSound(true); orderShuffled.find(o => o.id === id).clicked = true; step4Node++; renderG4(); } 
      else { 
        playSound(false); applyDamage(20); const c = document.getElementById('g4-container'); c.classList.add('bg-red-900/30'); 
        setTimeout(() => { c.classList.remove('bg-red-900/30'); initG4(); }, 500); 
      }
      saveCardProgress();
    }

    let step5 = 0, currentOptsG5 = [];
    function renderG5() {
      if(blockTimer) { clearInterval(blockTimer); blockTimer = null; }
      if(step5 >= oddData.length) { gamesStatus.g5 = true; updateHUD(); calcFinalScore(); return document.getElementById('g5-container').innerHTML = `<div class="text-center p-6"><div class="text-5xl mb-3">🕵️</div><div class="text-purple-400 font-black text-xl">AMEAÇAS IDENTIFICADAS!</div></div>`; }
      const q = oddData[step5];
      currentOptsG5 = shuffle(q.options);
      const c = document.getElementById('g5-container');
      c.innerHTML = `<div class="flex justify-between items-center mb-3"><span class="text-xs font-bold text-purple-400 bg-purple-400/10 px-2 py-1 rounded">Infiltração ${step5 + 1}/${oddData.length}</span></div><h4 class="text-sm text-white font-bold mb-4">${q.question}</h4><div id="g5-opts" class="grid grid-cols-1 sm:grid-cols-2 gap-3"></div><div id="fb5" class="hidden mt-4 p-4 rounded-lg font-bold text-sm"></div>`;
      const optDiv = document.getElementById('g5-opts');
      currentOptsG5.forEach((opt, i) => { 
        const btn = document.createElement('button');
        btn.className = 'p-4 bg-slate-800 border border-slate-700 rounded-lg text-sm hover:bg-slate-700 font-bold text-gray-200 transition-colors g5-btn flex items-center';
        btn.innerHTML = `<span class="mr-3 font-bold text-purple-400 bg-purple-900/30 px-2 py-0.5 rounded border border-purple-500/20">${i+1}</span> <span>${opt.text}</span>`;
        btn.onclick = () => checkG5(i);
        optDiv.appendChild(btn);
      });
    }
    function checkG5(idx) {
      const opt = currentOptsG5[idx];
      document.querySelectorAll('.g5-btn').forEach(b => b.disabled = true);
      const el = document.getElementById(`fb5`); el.classList.remove('hidden');
      playSound(opt.isOdd);
      if(opt.isOdd) { 
        hitsG5++;
        el.className = 'mt-4 p-4 rounded-lg font-bold text-sm bg-emerald-900/60 text-emerald-300 border border-emerald-500/30'; 
        el.innerHTML = `✅ INTRUSO IDENTIFICADO!<br><br><strong>Explicação:</strong> ${opt.explanation}`;
        const nxt = document.createElement('button'); nxt.className = 'block mt-3 bg-emerald-600 px-4 py-2 rounded text-white text-xs shortcut-next'; nxt.innerText = 'Próxima ➔ (Espaço)'; nxt.onclick = () => { step5++; renderG5(); };
        el.appendChild(nxt);
      } else { 
        el.className = 'mt-4 p-4 rounded-lg font-bold text-sm bg-red-900/60 text-red-300 border border-red-500/30'; 
        const actualOdd = currentOptsG5.find(o => o.isOdd);
        el.innerHTML = `❌ OPÇÃO LEGÍTIMA!<br><br>Sua escolha (${opt.text}) é válida: <strong>${opt.explanation}</strong><br><br>👉 O Infiltrado era: <strong>${actualOdd.text}</strong><br>Motivo: ${actualOdd.explanation}`;
        applyDamage(20);
        triggerPunishment('g5-container', 5, () => { step5++; renderG5(); });
      }
      saveCardProgress();
    }

    document.addEventListener("DOMContentLoaded", () => {
      loadCardProgress();
      try {
        if (typeof renderMathInElement === 'function') {
          renderMathInElement(document.body, {
            delimiters: [
              { left: "$$", right: "$$", display: true },
              { left: "$", right: "$", display: false }
            ],
            throwOnError: false,
            errorColor: '#f87171'
          });
        }
      } catch (e) { console.warn(e); }

      try {
        if (typeof questionsData !== 'undefined' && questionsData.length > 0) renderG1();
        if (typeof matchData !== 'undefined' && matchData.length > 0) initG2();
        if (typeof tfData !== 'undefined' && tfData.length > 0) initG3();
        if (typeof orderData !== 'undefined' && orderData.length > 0) initG4();
        if (typeof oddData !== 'undefined' && oddData.length > 0) renderG5();
      } catch (e) { console.error(e); }

      document.addEventListener('keydown', (e) => {
        if (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA') return;
        if (e.code === 'Space') {
          const nextBtn = document.querySelector('.game-view:not(.hidden) .shortcut-next');
          if (nextBtn) { e.preventDefault(); nextBtn.click(); return; }
        }
        if (e.key === ',' || e.key === '.') {
          const tabs = Array.from(document.querySelectorAll('.tab-btn'));
          const activeIdx = tabs.findIndex(t => t.classList.contains('active'));
          if (activeIdx >= 0) {
            let nxtIdx = e.key === '.' ? activeIdx + 1 : activeIdx - 1;
            if (nxtIdx >= tabs.length) nxtIdx = 0;
            if (nxtIdx < 0) nxtIdx = tabs.length - 1;
            tabs[nxtIdx].click();
          }
        }
        if (!document.getElementById('game-tf').classList.contains('hidden')) {
          if (e.key.toLowerCase() === 'v') { const b = document.getElementById('btn-t'); if (b && !b.disabled) b.click(); }
          if (e.key.toLowerCase() === 'f') { const b = document.getElementById('btn-f'); if (b && !b.disabled) b.click(); }
        }
        if (e.key >= '1' && e.key <= '5') {
          const idx = parseInt(e.key) - 1;
          if (!document.getElementById('game-timeline').classList.contains('hidden')) {
            const btns = document.querySelectorAll('.g1-btn:not([disabled])');
            if (btns[idx]) btns[idx].click();
          }
          if (!document.getElementById('game-odd').classList.contains('hidden')) {
            const btns = document.querySelectorAll('.g5-btn:not([disabled])');
            if (btns[idx]) btns[idx].click();
          }
        }
      });
    });
