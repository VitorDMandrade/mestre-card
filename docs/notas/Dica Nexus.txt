```typescript
let constellationDroneAudio: HTMLAudioElement | null = null;

/** Inicia o Drone de Foco Cósmico em loop contínuo com fade-in suave */
export function startConstellationDrone(enabled = true, volume = 0.22) {
  if (!enabled || typeof window === 'undefined') return;
  try {
    if (!constellationDroneAudio) {
      constellationDroneAudio = new Audio('/assets/audio/tactical_focus_drone.mp3');
      constellationDroneAudio.loop = true;
    }
    constellationDroneAudio.volume = volume;
    constellationDroneAudio.play().catch(() => {});
  } catch (e) {
    console.warn('[AUDIO]: Drone indisponível:', e);
  }
}

/** Pausa o Drone Cósmico imediatamente */
export function stopConstellationDrone() {
  if (constellationDroneAudio) {
    constellationDroneAudio.pause();
    constellationDroneAudio.currentTime = 0;
  }
}

/** Ajusta dinamicamente o volume do Drone */
export function setConstellationDroneVolume(volume: number) {
  if (constellationDroneAudio) {
    constellationDroneAudio.volume = Math.max(0, Math.min(1, volume));
  }

/** Hover estelar: Chime harmônico de alta frequência (880Hz -> 1320Hz) */
export function playConstellationHoverSound(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  const osc = ctx.createOscillator();
  const gain = ctx.createGain();
  const filter = ctx.createBiquadFilter();

  osc.type = 'sine';
  osc.frequency.setValueAtTime(880, now);
  osc.frequency.exponentialRampToValueAtTime(1320, now + 0.12);

  filter.type = 'bandpass';
  filter.frequency.setValueAtTime(1100, now);
  filter.Q.setValueAtTime(4.0, now);

  gain.gain.setValueAtTime(0.001, now);
  gain.gain.linearRampToValueAtTime(0.09, now + 0.02);
  gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.18);

  osc.connect(filter);
  filter.connect(gain);
  gain.connect(ctx.destination);

  osc.start(now);
  osc.stop(now + 0.18);
}

/** Seleção gravitacional: Pulso ressonante de engate orbital */
export function playConstellationSelectSound(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  const osc = ctx.createOscillator();
  const gain = ctx.createGain();
  osc.type = 'triangle';
  osc.frequency.setValueAtTime(160, now);
  osc.frequency.exponentialRampToValueAtTime(520, now + 0.16);

  gain.gain.setValueAtTime(0.01, now);
  gain.gain.linearRampToValueAtTime(0.24, now + 0.03);
  gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.28);

  osc.connect(gain);
  gain.connect(ctx.destination);

  osc.start(now);
  osc.stop(now + 0.28);
}

/** Salto Hiperespacial (Warp): Abertura do card com varredura ascendente */
export function playConstellationWarpSound(enabled = true) {
  if (!enabled) return;
  const ctx = getAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;

  // Ruído com filtro passa-faixa dinâmico
  const bufferSize = Math.floor(ctx.sampleRate * 0.35);
  const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
  const data = buffer.getChannelData(0);
  for (let i = 0; i < bufferSize; i++) {
    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.4));
  }
  const noise = ctx.createBufferSource();
  noise.buffer = buffer;

  const filter = ctx.createBiquadFilter();
  filter.type = 'bandpass';
  filter.frequency.setValueAtTime(200, now);
  filter.frequency.exponentialRampToValueAtTime(3200, now + 0.35);

  const gain = ctx.createGain();
  gain.gain.setValueAtTime(0.01, now);
  gain.gain.linearRampToValueAtTime(0.3, now + 0.1);
  gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.35);

  noise.connect(filter);
  filter.connect(gain);
  gain.connect(ctx.destination);

  noise.start(now);
}
```

### 🌌 3. MOTOR GRÁFICO NATIVO A 60 FPS (`src/lib/constellation-engine.ts`)

Motor em Canvas puro sem Three.js (inicialização ultrarrápida, zero vazamento de memória). Gera o universo em camadas, processa conexões de laser em feixe senoidal e controla ondas de choque de busca:

```typescript
import { ACERVO_CATALOG, AcervoItem } from '../data/acervo-catalog';

export type NodeType = 'hub' | 'card' | 'axis' | 'banca' | 'pdf';

export interface ConstellationNode {
  id: string;
  type: NodeType;
  label: string;
  subLabel?: string;
  x: number;
  y: number;
  baseX: number;
  baseY: number;
  radius: number;
  color: string;
  pulsePhase: number;
  discipline?: string;
  data?: any; // Referência ao card ou AcervoItem
  connectedTo: string[]; // IDs dos nós ligados
}

export interface ConstellationEdge {
  from: string;
  to: string;
  color: string;
  weight: number;
}

export interface StarParticle {
  x: number;
  y: number;
  size: number;
  alpha: number;
  speed: number;
}

export interface SupernovaRing {
  x: number;
  y: number;
  radius: number;
  maxRadius: number;
  color: string;
  alpha: number;
}

const HUB_COLORS: Record<string, string> = {
  quimica: '#00f5ff',
  biologia: '#10b981',
  fisica: '#f59e0b',
  matematica: '#8b5cf6',
  humanas: '#f43f5e',
  geral: '#38bdf8'
};

export class ConstellationEngine {
  nodes: ConstellationNode[] = [];
  edges: ConstellationEdge[] = [];
  stars: StarParticle[] = [];
  supernovas: SupernovaRing[] = [];
  hoveredNodeId: string | null = null;
  selectedNodeId: string | null = null;

  // Câmera & Projeção
  cameraX = 0;
  cameraY = 0;
  zoom = 1;
  targetZoom = 1;

  constructor(cards: any[] = []) {
    this.buildConstellation(cards);
    this.initStarfield();
  }

  buildConstellation(cards: any[]) {
    this.nodes = [];
    this.edges = [];

    // 1. Núcleos Disciplinares Principais (Planetas-Mãe)
    const disciplines = [
      { id: 'hub-quimica', name: 'Química', disc: 'quimica', x: -450, y: -250 },
      { id: 'hub-biologia', name: 'Biologia', disc: 'biologia', x: 450, y: -250 },
      { id: 'hub-fisica', name: 'Física', disc: 'fisica', x: -500, y: 350 },
      { id: 'hub-matematica', name: 'Matemática', disc: 'matematica', x: 500, y: 350 },
      { id: 'hub-humanas', name: 'Humanas & História', disc: 'humanas', x: 0, y: -450 }
    ];

    disciplines.forEach(d => {
      this.nodes.push({
        id: d.id,
        type: 'hub',
        label: d.name,
        x: d.x,
        y: d.y,
        baseX: d.x,
        baseY: d.y,
        radius: 28,
        color: HUB_COLORS[d.disc],
        pulsePhase: Math.random() * Math.PI * 2,
        discipline: d.disc,
        connectedTo: []
      });
    });

    // 2. Estações das Bancas de Elite
    const bancas = [
      { id: 'banca-einstein', name: 'Albert Einstein Medicina', badge: '🩺 EINSTEIN', x: 180, y: 50, color: '#00f5ff' },
      { id: 'banca-enem', name: 'Exame Nacional (ENEM)', badge: '🌐 ENEM', x: -180, y: 50, color: '#f59e0b' },
      { id: 'banca-unesp', name: 'Vestibular UNESP', badge: '🏛️ UNESP', x: 0, y: 220, color: '#3b82f6' }
    ];

    bancas.forEach(b => {
      this.nodes.push({
        id: b.id,
        type: 'banca',
        label: b.badge,
        subLabel: b.name,
        x: b.x,
        y: b.y,
        baseX: b.x,
        baseY: b.y,
        radius: 20,
        color: b.color,
        pulsePhase: Math.random() * Math.PI * 2,
        connectedTo: []
      });
    });

    // 3. Nós de MestreCards cadastrados
    cards.forEach((card, idx) => {
      const cardId = `card-${card.id || idx}`;
      const disc = (card.subject || card.discipline || 'quimica').toLowerCase();
      const parentHub = this.nodes.find(n => n.id === `hub-${disc}`) || this.nodes[0];

      // Disposição orbital ao redor do Hub pai
      const angle = (idx / Math.max(1, cards.length)) * Math.PI * 2 + Math.random() * 0.4;
      const dist = 140 + (idx % 3) * 45;
      const x = parentHub.baseX + Math.cos(angle) * dist;
      const y = parentHub.baseY + Math.sin(angle) * dist;

      const cardColor = HUB_COLORS[disc] || '#38bdf8';

      this.nodes.push({
        id: cardId,
        type: 'card',
        label: card.title || `Card #${idx + 1}`,
        subLabel: card.topic || 'Conceito Fundamental',
        x,
        y,
        baseX: x,
        baseY: y,
        radius: 14,
        color: cardColor,
        pulsePhase: Math.random() * Math.PI * 2,
        discipline: disc,
        data: card,
        connectedTo: [parentHub.id]
      });

      parentHub.connectedTo.push(cardId);
      this.edges.push({ from: parentHub.id, to: cardId, color: cardColor, weight: 1.5 });

      // Conexão do card com a banca correspondente
      const targetBanca = idx % 2 === 0 ? 'banca-einstein' : 'banca-enem';
      const bancaNode = this.nodes.find(n => n.id === targetBanca);
      if (bancaNode) {
        bancaNode.connectedTo.push(cardId);
        this.nodes.find(n => n.id === cardId)?.connectedTo.push(bancaNode.id);
        this.edges.push({ from: cardId, to: bancaNode.id, color: 'rgba(255, 255, 255, 0.25)', weight: 0.8 });
      }
    });

    // 4. Satélites do Acervo Oficial (Top PDFs mais relevantes)
    const featuredPdfs = (ACERVO_CATALOG || []).slice(0, 35);
    featuredPdfs.forEach((pdf, pIdx) => {
      const pdfId = `pdf-${pdf.id || pIdx}`;
      const disc = pdf.discipline || 'geral';
      const parentHub = this.nodes.find(n => n.id === `hub-${disc}`) || this.nodes[0];

      const angle = (pIdx / featuredPdfs.length) * Math.PI * 2;
      const dist = 240 + (pIdx % 4) * 35;
      const x = parentHub.baseX + Math.cos(angle) * dist;
      const y = parentHub.baseY + Math.sin(angle) * dist;

      this.nodes.push({
        id: pdfId,
        type: 'pdf',
        label: pdf.title.length > 22 ? pdf.title.slice(0, 22) + '...' : pdf.title,
        subLabel: `${pdf.banca || disc.toUpperCase()} • ${pdf.sizeFormatted}`,
        x,
        y,
        baseX: x,
        baseY: y,
        radius: 7,
        color: 'rgba(255, 255, 255, 0.75)',
        pulsePhase: Math.random() * Math.PI * 2,
        discipline: disc,
        data: pdf,
        connectedTo: [parentHub.id]
      });

      parentHub.connectedTo.push(pdfId);
      this.edges.push({ from: parentHub.id, to: pdfId, color: 'rgba(255, 255, 255, 0.12)', weight: 0.5 });
    });
  }

  initStarfield() {
    this.stars = [];
    for (let i = 0; i < 220; i++) {
      this.stars.push({
        x: (Math.random() - 0.5) * 3000,
        y: (Math.random() - 0.5) * 3000,
        size: Math.random() * 1.8 + 0.4,
        alpha: Math.random() * 0.7 + 0.2,
        speed: Math.random() * 0.002 + 0.001
      });
    }
  }

  triggerSupernova(query: string) {
    if (!query || query.trim().length < 2) return;
    const q = query.toLowerCase();

    this.nodes.forEach(n => {
      const match = n.label.toLowerCase().includes(q) || (n.subLabel && n.subLabel.toLowerCase().includes(q));
      if (match) {
        this.supernovas.push({
          x: n.x,
          y: n.y,
          radius: n.radius,
          maxRadius: n.radius * 5.5,
          color: n.color,
          alpha: 0.9
        });
      }
    });
  }

  update(time: number) {
    // Oscilação suave dos nós no éter cósmico
    this.nodes.forEach(n => {
      n.x = n.baseX + Math.sin(time * 0.001 + n.pulsePhase) * 4.5;
      n.y = n.baseY + Math.cos(time * 0.0012 + n.pulsePhase) * 4.5;
    });

    // Animação das supernovas
    this.supernovas.forEach(sn => {
      sn.radius += (sn.maxRadius - sn.radius) * 0.08;
      sn.alpha *= 0.92;
    });
    this.supernovas = this.supernovas.filter(sn => sn.alpha > 0.04);

    // Zoom com amortecimento
    this.zoom += (this.targetZoom - this.zoom) * 0.1;
  }

  render(ctx: CanvasRenderingContext2D, width: number, height: number, filterType = 'all') {
    ctx.clearRect(0, 0, width, height);

    ctx.save();
    ctx.translate(width / 2 + this.cameraX, height / 2 + this.cameraY);
    ctx.scale(this.zoom, this.zoom);

    // 1. Estrelas de Fundo com Paralaxe
    this.stars.forEach(s => {
      ctx.fillStyle = `rgba(255, 255, 255, ${s.alpha})`;
      ctx.beginPath();
      ctx.arc(s.x, s.y, s.size, 0, Math.PI * 2);
      ctx.fill();
    });

    // 2. Ondas de Choque Supernova
    this.supernovas.forEach(sn => {
      ctx.strokeStyle = sn.color;
      ctx.lineWidth = 2.5;
      ctx.globalAlpha = sn.alpha;
      ctx.beginPath();
      ctx.arc(sn.x, sn.y, sn.radius, 0, Math.PI * 2);
      ctx.stroke();
      ctx.globalAlpha = 1;
    });

    const activeNode = this.hoveredNodeId
      ? this.nodes.find(n => n.id === this.hoveredNodeId)
      : this.selectedNodeId
      ? this.nodes.find(n => n.id === this.selectedNodeId)
      : null;

    // 3. Feixes de Laser / Linhas de Conexão
    this.edges.forEach(e => {
      const fromNode = this.nodes.find(n => n.id === e.from);
      const toNode = this.nodes.find(n => n.id === e.to);
      if (!fromNode || !toNode) return;

      const isHighlighted = activeNode && (activeNode.id === fromNode.id || activeNode.id === toNode.id);
      const isDimmed = activeNode && !isHighlighted;

      ctx.beginPath();
      ctx.moveTo(fromNode.x, fromNode.y);
      ctx.lineTo(toNode.x, toNode.y);

      if (isHighlighted) {
        ctx.strokeStyle = '#00f5ff';
        ctx.lineWidth = 2.5;
        ctx.shadowColor = '#00f5ff';
        ctx.shadowBlur = 10;
      } else {
        ctx.strokeStyle = isDimmed ? 'rgba(255, 255, 255, 0.04)' : e.color;
        ctx.lineWidth = e.weight;
        ctx.shadowBlur = 0;
      }
      ctx.stroke();
      ctx.shadowBlur = 0;
    });

    // 4. Renderização dos Nós Estelares
    this.nodes.forEach(n => {
      if (filterType !== 'all') {
        if (filterType === 'cards' && n.type !== 'card') return;
        if (filterType === 'bancas' && n.type !== 'banca') return;
        if (filterType === 'pdfs' && n.type !== 'pdf') return;
      }

      const isHovered = n.id === this.hoveredNodeId;
      const isSelected = n.id === this.selectedNodeId;
      const isConnected = activeNode && (activeNode.id === n.id || activeNode.connectedTo.includes(n.id));
      const isDimmed = activeNode && !isConnected;

      ctx.globalAlpha = isDimmed ? 0.2 : 1;

      // Halo Luminoso Pulsante
      if (isHovered || isSelected || n.type === 'hub') {
        ctx.beginPath();
        ctx.arc(n.x, n.y, n.radius * 1.8, 0, Math.PI * 2);
        ctx.fillStyle = n.color;
        ctx.globalAlpha = isHovered ? 0.35 : 0.15;
        ctx.fill();
        ctx.globalAlpha = isDimmed ? 0.2 : 1;
      }

      // Corpo Central do Nó
      ctx.beginPath();
      ctx.arc(n.x, n.y, n.radius, 0, Math.PI * 2);
      ctx.fillStyle = n.color;
      ctx.shadowColor = n.color;
      ctx.shadowBlur = isHovered || isSelected ? 18 : 8;
      ctx.fill();
      ctx.shadowBlur = 0;

      // Borda Externa Metálica
      ctx.strokeStyle = isSelected ? '#ffffff' : 'rgba(255, 255, 255, 0.4)';
      ctx.lineWidth = isSelected ? 2.5 : 1;
      ctx.stroke();

      // Rótulo de Texto Tático
      if (this.zoom > 0.65 || n.type === 'hub' || isHovered || isSelected) {
        ctx.fillStyle = isSelected ? '#ffffff' : isHovered ? '#00f5ff' : 'rgba(255, 255, 255, 0.85)';
        ctx.font = n.type === 'hub' ? 'bold 13px monospace' : '10px monospace';
        ctx.textAlign = 'center';
        ctx.fillText(n.label, n.x, n.y + n.radius + 13);
      }
    });

    ctx.globalAlpha = 1;
    ctx.restore();
  }

  // Raycasting 2D para Detecção de Clique / Hover
  getNodeAt(screenX: number, screenY: number, width: number, height: number): ConstellationNode | null {
    const worldX = (screenX - width / 2 - this.cameraX) / this.zoom;
    const worldY = (screenY - height / 2 - this.cameraY) / this.zoom;

    for (let i = this.nodes.length - 1; i >= 0; i--) {
      const n = this.nodes[i];
      const dist = Math.hypot(n.x - worldX, n.y - worldY);
      if (dist <= n.radius + 6) {
        return n;
      }
    }
    return null;
  }
}
```

```tsx
import React, { useRef, useEffect, useState, useMemo } from 'react';
import { ConstellationEngine, ConstellationNode } from '../../lib/constellation-engine';
import { 
  startConstellationDrone, 
  stopConstellationDrone, 
  playConstellationHoverSound, 
  playConstellationSelectSound, 
  playConstellationWarpSound 
} from '../../lib/audio';

interface ConstellationViewProps {
  cards: any[];
  onSelectCard: (card: any) => void;
  onBackToDashboard: () => void;
  soundEnabled?: boolean;
}

export const ConstellationView: React.FC<ConstellationViewProps> = ({
  cards,
  onSelectCard,
  onBackToDashboard,
  soundEnabled = true
}) => {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const engineRef = useRef<ConstellationEngine | null>(null);

  const [searchQuery, setSearchQuery] = useState('');
  const [filterType, setFilterType] = useState<'all' | 'cards' | 'bancas' | 'pdfs'>('all');
  const [dronePlaying, setDronePlaying] = useState(true);
  const [selectedNode, setSelectedNode] = useState<ConstellationNode | null>(null);

  const isDragging = useRef(false);
  const lastMousePos = useRef({ x: 0, y: 0 });

  useEffect(() => {
    const engine = new ConstellationEngine(cards);
    engineRef.current = engine;

    if (soundEnabled && dronePlaying) {
      startConstellationDrone(true, 0.2);
    }

    let animationFrameId: number;
    const renderLoop = (time: number) => {
      const canvas = canvasRef.current;
      if (canvas && engineRef.current) {
        const ctx = canvas.getContext('2d');
        if (ctx) {
          engineRef.current.update(time);
          engineRef.current.render(ctx, canvas.width, canvas.height, filterType);
        }
      }
      animationFrameId = requestAnimationFrame(renderLoop);
    };

    animationFrameId = requestAnimationFrame(renderLoop);

    const handleResize = () => {
      if (canvasRef.current) {
        canvasRef.current.width = window.innerWidth;
        canvasRef.current.height = window.innerHeight;
      }
    };
    handleResize();
    window.addEventListener('resize', handleResize);

    return () => {
      cancelAnimationFrame(animationFrameId);
      window.removeEventListener('resize', handleResize);
      stopConstellationDrone();
    };
  }, [cards]);

  const handleSearchChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const q = e.target.value;
    setSearchQuery(q);
    if (engineRef.current) {
      engineRef.current.triggerSupernova(q);
    }
  };

  const toggleDrone = () => {
    if (dronePlaying) {
      stopConstellationDrone();
      setDronePlaying(false);
    } else {
      startConstellationDrone(soundEnabled, 0.2);
      setDronePlaying(true);
    }
  };

  const handleMouseDown = (e: React.MouseEvent) => {
    isDragging.current = true;
    lastMousePos.current = { x: e.clientX, y: e.clientY };
  };

  const handleMouseMove = (e: React.MouseEvent) => {
    const canvas = canvasRef.current;
    const engine = engineRef.current;
    if (!canvas || !engine) return;

    if (isDragging.current) {
      const dx = e.clientX - lastMousePos.current.x;
      const dy = e.clientY - lastMousePos.current.y;
      engine.cameraX += dx;
      engine.cameraY += dy;
      lastMousePos.current = { x: e.clientX, y: e.clientY };
    } else {
      const rect = canvas.getBoundingClientRect();
      const node = engine.getNodeAt(e.clientX - rect.left, e.clientY - rect.top, canvas.width, canvas.height);
      if (node && node.id !== engine.hoveredNodeId) {
        engine.hoveredNodeId = node.id;
        playConstellationHoverSound(soundEnabled);
      } else if (!node && engine.hoveredNodeId) {
        engine.hoveredNodeId = null;
      }
    }
  };

  const handleMouseUp = () => {
    isDragging.current = false;
  };

  const handleClick = (e: React.MouseEvent) => {
    const canvas = canvasRef.current;
    const engine = engineRef.current;
    if (!canvas || !engine) return;

    const rect = canvas.getBoundingClientRect();
    const node = engine.getNodeAt(e.clientX - rect.left, e.clientY - rect.top, canvas.width, canvas.height);
    if (node) {
      engine.selectedNodeId = node.id;
      setSelectedNode(node);
      playConstellationSelectSound(soundEnabled);
    }
  };

  const handleWheel = (e: React.WheelEvent) => {
    e.preventDefault();
    if (!engineRef.current) return;
    const zoomFactor = e.deltaY < 0 ? 1.12 : 0.88;
    engineRef.current.targetZoom = Math.max(0.35, Math.min(2.8, engineRef.current.targetZoom * zoomFactor));
```
